bash -c 'set -e
PROFILE='\''profiles/it'\''\'\'''\''s a; touch pwned #'\''
mkdir -p "$PROFILE" "$HOME/.tiny-agents"
PROFILE="$(cd "$PROFILE" && pwd)"
exec 8>"$HOME/.tiny-agents/browser-launch.lock"
flock -w 25 8 || { echo "Gave up after 25s waiting for another browser to finish starting on the VM" >&2; exit 1; }
if [ -s "$PROFILE/.stack-pids" ]; then xargs -r kill < "$PROFILE/.stack-pids" 2>/dev/null || true; fi
pkill -f -- "--user-data-dir=$PROFILE( |\$)" 2>/dev/null || true
TA_STOP=$((SECONDS + 10))
while pgrep -f -- "--user-data-dir=$PROFILE( |\$)" >/dev/null 2>&1 && [ "$SECONDS" -lt "$TA_STOP" ]; do sleep 0.2; done
sleep 0.3
PORTS="$(python3 -c '\''import socket
def free():
    s = socket.socket()
    s.bind(("127.0.0.1", 0))
    p = s.getsockname()[1]
    s.close()
    return p
print(free(), free())'\'')"
set -- $PORTS
VNC=$1
CDP=$2
DISP=""
for n in $(seq 90 189); do
  [ -e "/tmp/.X11-unix/X$n" ] && continue
  [ -e "/tmp/.X$n-lock" ] && continue
  DISP="$n"; break
done
[ -n "$DISP" ] || { echo "No free X display on the VM" >&2; exit 1; }
XLOG="$PROFILE/.xvnc.log"
CLOG="$PROFILE/.chrome.log"
rm -f "$PROFILE/.stack-pids"
setsid nohup Xtigervnc ":$DISP" -geometry 1024x767 -depth 24 -rfbport "$VNC" -localhost -nolisten tcp -SecurityTypes None -AlwaysShared -desktop tiny-agents </dev/null >"$XLOG" 2>&1 8>&- &
XPID=$!
echo "$XPID" > "$PROFILE/.stack-pids"
XWAIT=$((SECONDS + 10))
while [ ! -S "/tmp/.X11-unix/X$DISP" ] && [ "$SECONDS" -lt "$XWAIT" ]; do sleep 0.1; done
DISPLAY=":$DISP" setsid nohup vncconfig -nowin </dev/null >"$PROFILE/.vncconfig.log" 2>&1 8>&- &
echo "$!" >> "$PROFILE/.stack-pids"
DISPLAY=":$DISP" setsid nohup google-chrome-stable --user-data-dir="$PROFILE" --remote-debugging-port="$CDP" --remote-debugging-address=127.0.0.1 --window-position=0,0 --window-size=1024,767 --force-device-scale-factor=1 --no-first-run --no-default-browser-check --password-store=basic --use-mock-keychain --disable-features=Translate,MediaRouter --disable-session-crashed-bubble --hide-crash-restore-bubble about:blank </dev/null >"$CLOG" 2>&1 8>&- &
CPID=$!
echo "$CPID" >> "$PROFILE/.stack-pids"
UP=0
CDPWAIT=$((SECONDS + 30))
while [ "$SECONDS" -lt "$CDPWAIT" ]; do
  curl -sf --max-time 2 "http://127.0.0.1:$CDP/json/version" >/dev/null 2>&1 && { UP=1; break; }
  sleep 0.2
done
if [ "$UP" != 1 ]; then
  echo "Chrome did not start on the VM within 30s" >&2
  tail -n 20 "$XLOG" "$CLOG" >&2 || true
  xargs -r kill < "$PROFILE/.stack-pids" 2>/dev/null || true
  rm -f "$PROFILE/.stack-pids"
  exit 1
fi
echo "ok display=$DISP vnc=$VNC cdp=$CDP"'
