bash -c 'PROFILE='\''.tiny-agents/computer-profiles/bot-1'\''
[ -f "$PROFILE/.desktop-env" ] || { echo "The desktop is not running" >&2; exit 97; }
. "$PROFILE/.desktop-env"
[ -S "/tmp/.X11-unix/X${DISPLAY#:}" ] || { echo "The desktop is not running" >&2; exit 97; }
WANT_CLASS='\''firefox'\''
WANT_TITLE='\''log in — it'\''\'\'''\''s $home'\''
WID="$(xdotool getactivewindow 2>/dev/null || true)"
[ -n "$WID" ] || { echo "no window" >&2; exit 96; }
HEX="$(printf "0x%08x" "$WID")"
CLASS="$(wmctrl -l -x | awk -v id="$HEX" '\''$1 == id { print $3; exit }'\'')"
TITLE="$(xdotool getwindowname "$WID" 2>/dev/null || true)"
LC_CLASS="${CLASS,,}"
LC_TITLE="${TITLE,,}"
MATCH=0
case ".$LC_CLASS." in *".$WANT_CLASS."*) MATCH=1 ;; esac
if [ "$MATCH" = 1 ] && [ -n "$WANT_TITLE" ]; then case "$LC_TITLE" in *"$WANT_TITLE"*) ;; *) MATCH=0 ;; esac; fi
[ "$MATCH" = 1 ] || { echo "$CLASS — $TITLE" >&2; exit 96; }
xdotool type --clearmodifiers --delay 12 --file -'
